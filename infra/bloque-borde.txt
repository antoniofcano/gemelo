# Bloque para el Caddyfile de BORDE (el de Oficultor, dueño de 80/443: ADR-0011). Se pega tal cual: SIN variables
# de entorno ni marcadores. El borde termina TLS (certificado automático) y pasa todo al Caddy interno del gemelo
# (servicio `web`, alias gemelo-web en la red externa `borde`), que pone las cabeceras de seguridad, la CSP, la caché
# y la autorización (infra/Caddyfile). Caddy no se fía del X-Forwarded-For que mande un cliente: pone el suyo.
# Lo NO probado contra el borde real: doc/operacion/RUNBOOK-VPS.md.
gemelo.canoydamas.es {
	# V7: la subida de una foto (POST /api/medios) admite ~9 MB; el Caddy interno vuelve a limitarla. Todo lo demás, 2 MB.
	handle /api/medios {
		request_body {
			max_size 9MB
		}
		reverse_proxy gemelo-web:8080
	}
	handle {
		request_body {
			max_size 2MB
		}
		reverse_proxy gemelo-web:8080
	}
}
